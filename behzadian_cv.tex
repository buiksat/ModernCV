\section{Research Profile}
Research scientist studying policy improvement when the model of the world or the evaluator is imperfect. My work spans robust MDPs under uncertain dynamics and policy improvement under truncated evaluators with verifier-style rewards, with applications to RL post-training and reasoning.

%------------------------------------------------------------
\section{Research Highlights}

\cvitem{}{%
\textbf{Policy Improvement under Imperfect Evaluators}\newline
\emph{\href{https://openreview.net/pdf?id=rIzREKws05}{UPI-TRM}, ICLR 2026 Workshop on AI with Recursive Self-Improvement}\newline
Derived finite-depth policy-improvement guarantees that separate Bellman residual from truncation bias, and studied policy improvement with verifier-style rewards under truncated evaluators.
}

\cvitem{}{%
\textbf{Robust and Risk-Sensitive Markov Decision Processes}\newline
\emph{NeurIPS 2021 and AISTATS 2021}\newline
Developed scalable algorithms for \(L_\infty\)-constrained S-rectangular robust MDPs and percentile criteria under model uncertainty.
}
